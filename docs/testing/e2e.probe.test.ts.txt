/**
 * AUDIT PROBE (throwaway worktree only): does what the API's planner + sponsored
 * send builder actually emit pass the phone's verifyKuruTrade?
 */
import assert from 'node:assert/strict';
import { test } from 'node:test';

import { KURU_TESTNET_MARKETS, KURU_TESTNET_TOKENS, KURU_TESTNET_CONTRACTS, NATIVE_TOKEN } from '@sente/venues/kuru';
import { getAddress, isAddressEqual, parseEther, type Address, type PublicClient } from 'viem';

import { PrivyClient } from '../../../../services/api/src/agents/privy/privy.client.ts';
import { planKuru } from '../../../../services/api/src/trade/kuru-planner.ts';
import {
  sponsoredCallTransaction,
  sponsoredSendBody,
  walletRpcPath,
} from '../../../../services/api/src/wallet/send/sponsored-send.ts';
import { depositCapAtoms, worstPriceUnits, type MarketFacts } from './kuruMarket.ts';
import type { KuruIntent, PreparedStep } from './types.ts';
import { verifyKuruTrade } from './verifyKuru.ts';

const WALLET = getAddress('0x1111111111111111111111111111111111111111');
const WALLET_ID = 'wallet00000000000000test';
const TRADE_ID = '5b0c8f1e-9a3d-4c2b-8e7f-1a2b3c4d5e6f';
const MON_USDC = KURU_TESTNET_MARKETS.find((m) => m.symbol === 'MON-USDC')!;
const USDC = KURU_TESTNET_TOKENS.USDC;
const ACCOUNT_CORE = KURU_TESTNET_CONTRACTS.accountCore;

type World = {
  tickSize?: bigint;
  kuruFree?: Record<string, bigint>;
  walletMon?: bigint;
  accountId?: bigint;
  liveOrderId?: bigint;
  bid?: number;
  ask?: number;
};
const TAKER = 700n;
const MAKER = 4_000n;

function fakeClient(world: World): PublicClient {
  const readContract = ({ address, functionName, args }: { address: Address; functionName: string; args?: readonly unknown[] }) => {
    if (functionName === 'getMarketParams') {
      const market = KURU_TESTNET_MARKETS.find((m) => isAddressEqual(m.address, address))!;
      return Promise.resolve([market.pricePrecision, market.sizePrecision, world.tickSize ?? 1n, 1_000_000n, 10n ** 30n, TAKER, MAKER]);
    }
    if (functionName === 'getBalance') {
      const token = (args![1] as Address).toLowerCase();
      return Promise.resolve(world.kuruFree?.[token] ?? 0n);
    }
    if (functionName === 'userRegistry') return Promise.resolve(world.accountId ?? 0n);
    if (functionName === 'getOrderId') return Promise.resolve(world.liveOrderId ?? 0n);
    if (functionName === 'bestBidAsk') return Promise.resolve([world.bid ?? 19_900, world.ask ?? 20_100]);
    return Promise.reject(new Error(`fake client: no ${functionName}`));
  };
  const getBalance = () => Promise.resolve(world.walletMon ?? parseEther('100'));
  return { readContract, getBalance } as unknown as PublicClient;
}

function facts(world: World): MarketFacts {
  return {
    params: {
      pricePrecision: MON_USDC.pricePrecision,
      sizePrecision: MON_USDC.sizePrecision,
      tickSize: world.tickSize ?? 1n,
      minQuoteNotional: 1_000_000n,
      maxQuoteNotional: 10n ** 30n,
      takerFeePps: TAKER,
      makerFeePps: MAKER,
    },
    bestBid: BigInt(world.bid ?? 19_900),
    bestAsk: BigInt(world.ask ?? 20_100),
  };
}

const privy = new PrivyClient({ appId: 'app-id-test', appSecret: 'secret' });

/** Exactly what TradeService.prepare + PrivyUserWalletProvider.prepareSend build. */
async function prepare(intent: KuruIntent, world: World, atomicBatch: boolean): Promise<PreparedStep[]> {
  const plan = await planKuru(intent as never, { client: fakeClient(world), wallet: WALLET, atomicBatch });
  return plan.steps.map((planned, index) => {
    const { to, data, value } = planned.transaction;
    const path = walletRpcPath(WALLET_ID);
    const body = sponsoredSendBody(sponsoredCallTransaction(to, data, value));
    const payload = privy.authorizationPayload('POST', path, body, { idempotencyKey: `sente-trade:${TRADE_ID}:${index}` });
    return { index, kind: planned.kind, title: planned.title, payload };
  });
}

function verify(steps: PreparedStep[], intent: KuruIntent, world: World, slippageBps = 50) {
  return verifyKuruTrade(steps, { walletId: WALLET_ID, wallet: WALLET, intent, facts: facts(world), slippageBps });
}

const SIZE = 500n * 10n ** 8n; // 500 MON
const usdcKey = USDC.address.toLowerCase();

for (const atomic of [false, true]) {
  test(`buy limit with USDC shortfall, atomic=${atomic}`, async () => {
    const world: World = { kuruFree: { [usdcKey]: 4_000_000n } };
    const cap = depositCapAtoms({ side: 'buy', price: 20_000n, quantity: SIZE, tif: 'gtc' }, facts(world).params, { quote: 6, base: 18 });
    const intent: KuruIntent = { kind: 'kuru.place', clientTradeId: TRADE_ID, market: MON_USDC.address, side: 'buy', orderType: 'limit', sizeAtoms: SIZE.toString(), priceUnits: '20000', maxDepositAtoms: cap.toString() };
    const steps = await prepare(intent, world, atomic);
    const r = verify(steps, intent, world);
    assert.deepEqual(r, { ok: true }, JSON.stringify({ r, kinds: steps.map((s) => s.kind) }));
  });

  test(`post-only buy limit, no shortfall, atomic=${atomic}`, async () => {
    const world: World = { kuruFree: { [usdcKey]: 10n ** 9n } };
    const intent: KuruIntent = { kind: 'kuru.place', clientTradeId: TRADE_ID, market: MON_USDC.address, side: 'buy', orderType: 'limit', postOnly: true, sizeAtoms: SIZE.toString(), priceUnits: '20000', maxDepositAtoms: '0' };
    const steps = await prepare(intent, world, atomic);
    assert.deepEqual(verify(steps, intent, world), { ok: true });
  });

  test(`buy market (IOC) with USDC shortfall at the taker rate, atomic=${atomic}`, async () => {
    const world: World = { ask: 20_100, tickSize: 1n };
    const worst = worstPriceUnits(20_100n, 50, 1n, 'buy');
    const cap = depositCapAtoms({ side: 'buy', price: worst, quantity: SIZE, tif: 'ioc' }, facts(world).params, { quote: 6, base: 18 });
    const intent: KuruIntent = { kind: 'kuru.place', clientTradeId: TRADE_ID, market: MON_USDC.address, side: 'buy', orderType: 'market', sizeAtoms: SIZE.toString(), priceUnits: worst.toString(), maxDepositAtoms: cap.toString() };
    const steps = await prepare(intent, world, atomic);
    assert.deepEqual(verify(steps, intent, world), { ok: true }, JSON.stringify(steps.map((s) => s.kind)));
  });

  test(`sell market funded with native MON, atomic=${atomic}`, async () => {
    const world: World = { bid: 19_900, tickSize: 1n, walletMon: parseEther('600') };
    const worst = worstPriceUnits(19_900n, 50, 1n, 'sell');
    const cap = depositCapAtoms({ side: 'sell', price: worst, quantity: SIZE, tif: 'ioc' }, facts(world).params, { quote: 6, base: 18 });
    const intent: KuruIntent = { kind: 'kuru.place', clientTradeId: TRADE_ID, market: MON_USDC.address, side: 'sell', orderType: 'market', sizeAtoms: SIZE.toString(), priceUnits: worst.toString(), maxDepositAtoms: cap.toString() };
    const steps = await prepare(intent, world, atomic);
    const r = verify(steps, intent, world);
    assert.deepEqual(r, { ok: true }, JSON.stringify({ r, kinds: steps.map((s) => s.kind) }));
  });

  test(`sell limit with MON already free, atomic=${atomic}`, async () => {
    const world: World = { kuruFree: { [NATIVE_TOKEN.toLowerCase()]: parseEther('500') }, walletMon: 0n };
    const intent: KuruIntent = { kind: 'kuru.place', clientTradeId: TRADE_ID, market: MON_USDC.address, side: 'sell', orderType: 'limit', sizeAtoms: SIZE.toString(), priceUnits: '20500', maxDepositAtoms: '0' };
    const steps = await prepare(intent, world, atomic);
    assert.deepEqual(verify(steps, intent, world), { ok: true });
  });

  test(`cancel, atomic=${atomic}`, async () => {
    const world: World = { accountId: 63n, liveOrderId: 3683n };
    const intent: KuruIntent = { kind: 'kuru.cancel', clientTradeId: TRADE_ID, market: MON_USDC.address, orderId: '3:3683' };
    const steps = await prepare(intent, world, atomic);
    assert.deepEqual(verify(steps, intent, world), { ok: true });
  });

  test(`withdraw, atomic=${atomic}`, async () => {
    const world: World = { kuruFree: { [usdcKey]: 5_000_000n } };
    const intent: KuruIntent = { kind: 'kuru.withdraw', clientTradeId: TRADE_ID, token: USDC.address, amountAtoms: '5000000' };
    const steps = await prepare(intent, world, atomic);
    assert.deepEqual(verify(steps, intent, world), { ok: true });
  });
}

test('probe sanity: a tampered plan is still refused', async () => {
  const world: World = { kuruFree: { [usdcKey]: 4_000_000n } };
  const cap = depositCapAtoms({ side: 'buy', price: 20_000n, quantity: SIZE, tif: 'gtc' }, facts(world).params, { quote: 6, base: 18 });
  const intent: KuruIntent = { kind: 'kuru.place', clientTradeId: TRADE_ID, market: MON_USDC.address, side: 'buy', orderType: 'limit', sizeAtoms: SIZE.toString(), priceUnits: '20000', maxDepositAtoms: cap.toString() };
  const steps = await prepare(intent, world, false);
  const r = verify(steps, { ...intent, priceUnits: '20001' }, world);
  assert.equal(r.ok, false);
  void ACCOUNT_CORE;
});
